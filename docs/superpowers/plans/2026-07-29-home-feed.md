> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 HomeFeed 主屏从「框架完整、数据稀疏」打磨成「四类结果性事件齐全 + Dock 常用直达 + 问候动态叙事 + Feed 已读状态」的完整 §4.2 体验。

```python
def count_unread(self) -> int:
    row = self._conn.execute("SELECT COUNT(*) FROM feed WHERE read=0").fetchone()
    return row[0] if row else 0

def mark_read(self, feed_id: int) -> bool:
    cur = self._conn.execute("UPDATE feed SET read=1 WHERE id=?", (feed_id,))

    cur = self._conn.execute("UPDATE feed SET read=1 WHERE read=0")
```

```python
def append_hourly(self, kind: str, text: str, meta: dict, hour_key: int) -> None:
    mtype = (meta or {}).get("type", "")
    try:
        row = self._conn.execute(
            "SELECT id, text FROM feed WHERE kind=? AND json_extract(meta,'$.type')=? "
            "AND json_extract(meta,'$.hour')=? ORDER BY ts DESC LIMIT 1",
            (kind, mtype, hour_key),
        ).fetchone()
        if row:
            self._conn.execute(
                "UPDATE feed SET text=?, ts=?, read=0 WHERE id=?",
                (f"{row[1]}；{text}", hour_key, row[0]),
            )
        else:
            self._conn.execute(
                "INSERT INTO feed (ts, kind, text, meta, read) VALUES (?,?,?,?,0)",
                (hour_key, kind, text, json.dumps(meta, ensure_ascii=False)),
            )
        self._conn.commit()
    except Exception as e:
        print(f"[yibao] feed.append_hourly 失败（已忽略）：{e}", file=sys.stderr)
```
（`hour_key` 同时用作 ts 与去重键；合并时把 read 重置为 0——有新内容视为未读。）

```python
    all_plugins = [{"id": p.id, "name": p.name} for p in <loaded_plugins>]
    unpinned_sorted = sorted(
        [p for p in all_plugins if p["id"] not in pinned],
        key=lambda p: counts.get(p["id"], 0), reverse=True,
    )
    dock = [{"id": pid, "name": _name_of(pid), "pinned": True} for pid in pinned[:5]]
    for p in unpinned_sorted:
        if len(dock) >= 5: break
        dock.append({"id": p["id"], "name": p["name"], "pinned": False})
    if not dock:  # 全空退化字母序前 5
        dock = [{"id": p["id"], "name": p["name"], "pinned": False} for p in sorted(all_plugins, key=lambda x: x["name"])[:5]]
    return dock
```
